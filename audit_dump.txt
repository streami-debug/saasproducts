async function startServer() {
  const app = express();
  const PORT = 3000;
  
  app.use(express.json());

  // API Routes
  
  // Set trust proxy so Auth.js can work behind reverse proxy correctly
  app.set('trust proxy', true);
  
  // Lazy Auth Route to prevent crashing on missing env vars during boot
  app.use("/api/auth/*", (req, res, next) => {
    if (!process.env.DATABASE_URL || !process.env.AUTH_SECRET) {
      return res.status(500).json({ error: "Missing required environment variables (DATABASE_URL or AUTH_SECRET)" });
    }

    const { Pool } = require("@neondatabase/serverless");
    const { PrismaNeon } = require("@prisma/adapter-neon");
    const { PrismaClient } = require("@prisma/client");
    
    // Create lazily (usually you'd memoize this in production)
    const connectionString = process.env.DATABASE_URL;
    const pool = new Pool({ connectionString });
    const adapter = new PrismaNeon(pool);
    const prisma = new PrismaClient({ adapter });

    return ExpressAuth({
      providers: [
        GitHub({
          clientId: process.env.GITHUB_ID,
          clientSecret: process.env.GITHUB_SECRET,
        })
      ],
      adapter: PrismaAdapter(prisma),
      secret: process.env.AUTH_SECRET,
      trustHost: true,
    })(req, res, next);
  });

  app.get("/api/templates", (req, res) => {
    let templates = MOCK_TEMPLATES;
    const category = req.query.category as string;
    const search = req.query.search?.toString().toLowerCase();

    if (category && category !== 'all') {
      templates = templates.filter(t => t.category === category);
    }
    if (search) {
      templates = templates.filter(t => 
        t.title.toLowerCase().includes(search) || 
        t.description.toLowerCase().includes(search) ||
        t.tags.some(tag => tag.toLowerCase().includes(search))
      );
    }

    // Filter by max/min price if requested
    const minPrice = parseInt(req.query.minPrice as string || '0');
    const maxPrice = parseInt(req.query.maxPrice as string || '1000');
    
    templates = templates.filter(t => t.price >= minPrice && t.price <= maxPrice);

    res.json({ templates });
  });

  app.get("/api/templates/:id", (req, res) => {
    const item = MOCK_TEMPLATES.find(t => t.id === req.params.id);
    if (item) {
      res.json(item);
    } else {
      res.status(404).json({ error: "Template not found" });
    }
  });

  app.post("/api/checkout/paypal", (req, res) => {
    // Generate a mock PayPal order ID
    res.json({ id: "PAYPAL_ORDER_" + Math.random().toString(36).substring(7) });
  });

  app.post("/api/checkout/capture", (req, res) => {
    // Verify and capture payment
    res.json({ success: true, url: "/dashboard/user?checkout=success" });
  });

  // Vite middleware for development
  if (process.env.NODE_ENV !== "production") {
    const vite = await createViteServer({
      server: { middlewareMode: true },
      appType: "spa",
    });
    app.use(vite.middlewares);
  } else {
    // Production asset serving
    const distPath = path.join(process.cwd(), 'dist');
    app.use(express.static(distPath));
    app.get('*', (req, res) => {
      res.sendFile(path.join(distPath, 'index.html'));
    });
  }

  app.listen(PORT, "0.0.0.0", () => {
    console.log(`Server running on http://0.0.0.0:${PORT}`);
  });
}

startServer();

CREATE TABLE templates (

    description TEXT NOT NULL,

    download_count_cache INTEGER DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL
);
